cmake_minimum_required(VERSION 3.25)
project(A125DigitalBass VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(CTest)
if(BUILD_TESTING)
  add_executable(a125_engine_tests tests/engine_tests.cpp)
  target_include_directories(a125_engine_tests PRIVATE src)
  add_test(NAME engine_sanity COMMAND a125_engine_tests)
endif()
if(EXISTS "third_party/vst3sdk/CMakeLists.txt")
  set(SMTG_ENABLE_VSTGUI_SUPPORT OFF CACHE BOOL "" FORCE)
  set(SMTG_ENABLE_VST3_PLUGIN_EXAMPLES OFF CACHE BOOL "" FORCE)
  set(SMTG_ENABLE_VST3_HOSTING_EXAMPLES OFF CACHE BOOL "" FORCE)
  set(SMTG_RUN_VST_VALIDATOR OFF CACHE BOOL "" FORCE)
  set(SMTG_CREATE_PLUGIN_LINK OFF CACHE BOOL "" FORCE)
  add_subdirectory(third_party/vst3sdk)
  smtg_add_vst3plugin(A125DigitalBass src/plugin.cpp src/bass_engine.hpp)
  target_include_directories(A125DigitalBass PRIVATE src)
  target_link_libraries(A125DigitalBass PRIVATE sdk)
  target_compile_features(A125DigitalBass PRIVATE cxx_std_23)
else()
  message(STATUS "SDK absent - compiling standalone DSP tests only")
endif()
